for data in courses_data:
    print(f"\n  [{data['department']}] {data['course_code']} — {data['course_name']}")

    course, course_created = Course.objects.get_or_create(
        course_code=data['course_code'],
        defaults={
            'course_name': data['course_name'],
            'department':  data['department'],
            'semester':    data['semester'],
            'is_active':   True,
        }
    )

    if course_created:
        created_courses += 1
        print(f"    Course created.")
    else:
        skipped_courses += 1
        print(f"    Course already exists — checking assignments...")

    for s in data['sections']:
        try:
            faculty = User.objects.get(email=s['faculty_email'])
        except User.DoesNotExist:
            print(f"    ERROR: Faculty not found — {s['faculty_email']} (run seed_users.py first)")
            continue

        assignment, a_created = CourseAssignment.objects.get_or_create(
            course=course,
            class_section=s['section'],
            defaults={
                'faculty':    faculty,
                'is_primary': s['is_primary'],
            }
        )

        if a_created:
            created_assignments += 1
            tag = ' [PRIMARY]' if s['is_primary'] else ''
            print(f"    Section {s['section']} -> {faculty.full_name}{tag}")
        else:
            skipped_assignments += 1
            print(f"    Section {s['section']} already assigned — skipped.")
